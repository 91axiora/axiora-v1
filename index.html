/ /* AXIORA ENERGY MAPPING */ const ENERGY = { 1: ["Pioneer", "Initiative and leadership"], 2: ["Harmony", "Balance and cooperation"], 3: ["Creator", "Creativity and expression"], 4: ["Builder", "Structure and stability"], 5: ["Explorer", "Freedom and exploration"], 6: ["Nurturer", "Care and responsibility"], 7: ["Seeker", "Knowledge and inner search"], 8: ["Achiever", "Achievement and power"], 9: ["Visionary", "Vision and higher purpose"] }; function calculate(){ const input = document.getElementById("nameInput").value .toUpperCase() .replace(/[^A-Z]/g,""); const error = document.getElementById("error"); const result = document.getElementById("result"); if(!input){ error.textContent = "Please enter a name."; result.style.display="none"; return; } let numbers=[]; let total=0; for(const letter of input){ if(AXIORA[letter]){ numbers.push(AXIORA[letter]); total += AXIORA[letter]; } } const energyNumber = reduceToSingleNumber(total); const energy = ENERGY[energyNumber]; document.getElementById("displayName") .textContent=input; document.getElementById("displayNumbers") .textContent=numbers.join(" + "); document.getElementById("displayTotal") .textContent="Total: " + total; document.getElementById("displayEnergy") .textContent= energyNumber + " — " + energy[0]; document.getElementById("displayEnergyDesc") .textContent=energy[1]; error.textContent=""; result.style.display="block"; } function reduceToSingleNumber(number){ while(number > 9){ number = String(number) .split("") .reduce( (sum,digit) => sum + Number(digit), 0 ); } return number; }
